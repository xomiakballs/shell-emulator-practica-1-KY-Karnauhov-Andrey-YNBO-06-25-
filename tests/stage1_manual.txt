# Команды для ручной проверки Этапа 1 (вводить в окне эмулятора).
# Справа после "=>" указан ожидаемый результат.

ls                 => ls: вызвана с аргументами: []
ls /home           => ls: вызвана с аргументами: ['/home']
cd /tmp            => cd: вызвана с аргументами: ['/tmp']
ls a b             => Ошибка: ls: слишком много аргументов
cd a b             => Ошибка: cd: слишком много аргументов
foobar 1 2         => Ошибка: неизвестная команда foobar
exit now           => Ошибка: exit: команда не принимает аргументов
exit               => Выход из эмулятора... (окно закрывается)
